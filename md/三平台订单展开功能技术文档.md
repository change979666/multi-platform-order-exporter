### 💻 核心实现代码

#### 展开函数 - expandCollapsedOrders()
```javascript
function expandCollapsedOrders() {
  return new Promise((resolve) => {
    try {
      // 获取所有展开按钮文本元素
      const toggles = Array.from(
        document.querySelectorAll('.order-item-expand .order-item-expand-text')
      );
      
      let clicked = 0;
      
      toggles.forEach((textEl) => {
        const txt = (textEl.textContent || '').trim();
        // 检查是否为折叠状态（包含"สินค้ารวม"）
        const isCollapsed = /สินค้ารวม/i.test(txt);
        
        if (isCollapsed) {
          // 找到可点击的父元素（通常是 .order-item-expand-inner）
          const inner = textEl.closest('.order-item-expand-inner') || textEl;
          // 模拟鼠标点击事件
          inner.dispatchEvent(new MouseEvent('click', { 
            bubbles: true, 
            cancelable: true 
          }));
          clicked += 1;
        }
      });
      
      // 如果有展开操作，等待DOM更新完成
      if (clicked > 0) {
        setTimeout(resolve, 400); // 等待400ms确保展开动画完成
      } else {
        resolve(); // 没有需要展开的订单，直接完成
      }
    } catch (error) {
      console.error('展开订单时发生错误:', error);
      resolve();
    }
  });
}
```

```javascript
    '.order-item-expand .order-item-expand-text'
  );

    const expandText = expandButton.textContent || '';
    // 检查是否包含泰语"สินค้ารวม"（共X项/件）

    }
  }
```

### 💻 核心实现代码

#### 主展开函数 - expandTkShowMoreButtons()
```javascript
function expandTkShowMoreButtons() {
  try {
    // 查找所有可能的按钮元素
    const allButtons = document.querySelectorAll(
      'button, div[role="button"], div[class*="button"], ' +
      'span[class*="button"], div, span, a'
    );
    
    // 过滤出展开/收起按钮
    const showMoreButtons = Array.from(allButtons).filter(btn => {
      const text = btn.textContent || '';
      return (text.includes('Show') && text.includes('more')) || 
             (text.includes('Hide') && text.includes('item'));
    });
    
    let foundButtons = false;
    let clickedCount = 0;
    const MAX_BUTTON_CLICKS = 20; // 安全限制
    
    showMoreButtons.forEach(button => {
      // 跳过已处理过的按钮
      if (button.dataset.processed === 'true') {
        return;
      }
      
      const buttonText = button.textContent || '';
      
      // 检查按钮文本
      const isExpandButton = 
        (buttonText.includes('Show') && 
         buttonText.includes('more') && 
         buttonText.includes('item')) ||
        (buttonText.includes('Hide') && 
         buttonText.includes('item'));
      
      if (isExpandButton) {
        // 找到按钮所在的订单行tr并添加边框
        const orderRow = button.closest('tr.goGcNp, tr.sc-dPFVfm, tr.fpLFQn');
        
        if (orderRow) {
          if (!orderRow.classList.contains('highlight-order-border')) {
            orderRow.classList.add('highlight-order-border');
          }
        }
        
        // 标记按钮已处理
        button.dataset.processed = 'true';
        
        // 为按钮添加高亮样式
        button.classList.add("highlight-expand-button");
        
        // 如果是展开按钮（包含Show和more），才执行点击操作
        if (buttonText.includes('Show') && buttonText.includes('more')) {
          if (clickedCount < MAX_BUTTON_CLICKS) {
            try {
              // 模拟点击展开按钮
              button.click();
              clickedCount++;
              foundButtons = true;
              
              // 展开后延迟重新高亮
              setTimeout(() => {
                highlightTkItems(); // 重新高亮新展开的商品
              }, 1000);
            } catch (clickError) {
              console.error('点击展开按钮失败:', clickError);
            }
          }
        }
      }
    });
    
    // 使用 MutationObserver 监听后续的DOM变化
    if (foundButtons) {
      const observer = new MutationObserver((mutations) => {
        const newButtons = document.querySelectorAll(
          '.sc-cTkwdZ.cTfdes:not([data-processed="true"])'
        );
        if (newButtons.length > 0) {
          observer.disconnect();
          setTimeout(() => {
            expandTkShowMoreButtons();
          }, 300);
        }
      });
      
      observer.observe(document.body, { 
        childList: true, 
        subtree: true 
      });
      
      // 3秒后自动停止观察
      setTimeout(() => {
        observer.disconnect();
      }, 3000);
    }
    
    return foundButtons;
  } catch (error) {
    console.warn('TK自动展开功能出错:', error.message);
    return false;
  }
}
```

#### 查找订单容器函数 - findOrderContainer()
```javascript
function findOrderContainer(element) {
  // 1. 先检查是否在表格行内（支持新旧版本）
  let container = element.closest('tr.goGcNp') || 
                 element.closest('tr.sc-dPFVfm') || 
                 element.closest('tr.fpLFQn');
  
  // 2. 如果不在表格行内，尝试查找订单卡片或容器
  if (!container) {
    container = element.closest('.order-item') || 
               element.closest('.order-container') || 
               element.closest('.order-card');
  }
  
  // 3. 如果还是没找到，尝试查找通用订单区域
  if (!container) {
    container = element.closest('section') || 
               element.closest('[id*="order"]') ||
               element.closest('[class*="order"]');
  }
  
  // 4. 最后尝试获取最近的大型容器
  if (!container) {
    const parents = [];
    let parent = element.parentElement;
    // 向上收集5层父元素
    for (let i = 0; parent && i < 5; i++) {
      parents.push(parent);
      parent = parent.parentElement;
    }
    // 优先选择有其他商品信息的大容器
    container = parents.find(p => 
      p.querySelector('[class*="product"]') || 
      p.querySelector('[class*="item"]') ||
      p.querySelector('img[src*="product"]')
    ) || parents[3] || parents[2];
  }
  
  return container;
}
```

```javascript
function isTikTokOrderPage() {
  try {
    // 1. 检查URL（最快）
    if (window.location.href.includes('tiktok.com/order') || 
        window.location.href.includes('tiktok.com/seller') ||
        window.location.href.includes('seller.tiktokglobalshop.com')) {
      return true;
    }
    
    // 2. 检查DOM特征
    const hasTkRows = document.querySelectorAll('tr.goGcNp').length > 0;
    if (hasTkRows) {
      return true;
    }
    
    // 3. 检查是否有Seller SKU文本
    const mainContent = document.querySelector('main') || document.body;
    const allDivs = mainContent.querySelectorAll('div');
    const divLimit = Math.min(allDivs.length, 100);
    
    for (let i = 0; i < divLimit; i++) {
      if (allDivs[i].textContent && 
          allDivs[i].textContent.includes('Seller SKU:')) {
        return true;
      }
    }
    
    return false;
  } catch (error) {
    console.warn('TK页面检测出错:', error);
    return false;
  }
}
```

```javascript
function applyHighlight() {
  try {
    // 防止重复执行
    if (document.body && 
        document.body.dataset.orderHighlightApplied === '1') {
      return;
    }
    
    // 检测平台类型
    const isTK = isTikTokOrderPage();
    
    if (isTK) {
      // TK平台处理逻辑
      expandTkShowMoreButtons();
      highlightTkItems();
      
      // 延迟再次处理
      setTimeout(() => {

        highlightTkItems();

        setTimeout(() => {
          highlightQuantities();
          highlightOrders();
          highlightSkuBrackets();

    console.error('applyHighlight error:', error);
  }
```

### 3. 移除高亮函数

#### removeHighlight()
```javascript
function removeHighlight() {
  // 检测平台类型
  if (isTikTokOrderPage()) {
    removeTkHighlights(); // TK平台专用移除函数
  } else {
    // 移除虾皮/Lazada平台高亮
    document.querySelectorAll('.highlight-item-amount')
      .forEach(el => el.classList.remove('highlight-item-amount'));
    
    document.querySelectorAll('.highlight-order-border')
      .forEach(el => el.classList.remove('highlight-order-border'));
    
    document.querySelectorAll('.highlight-expand-button')
      .forEach(el => el.classList.remove('highlight-expand-button'));
    
    document.querySelectorAll('span.highlight-sku')
      .forEach(span => {
        const text = document.createTextNode(span.textContent || '');
        span.parentNode && span.parentNode.replaceChild(text, span);
      });
  }
  
  // 清除执行标记
  if (document.body) {
    delete document.body.dataset.orderHighlightApplied;
  }
}
```

### 📊 选择器对比表

| 功能 | 虾皮（Shopee） | Lazada | TikTok Shop |
|------|---------------|---------|-------------|
| **展开按钮** | `.order-item-expand .order-item-expand-text` | 同虾皮 + `.expand-trigger` | `button`包含"Show more"文本 |

| **商品数量** | `.item-amount` | `.order-item-count` | `.isUnTN`, `.iqakNZ` |
| **识别文本** | 泰语"สินค้ารวม" | 泰语"สินค้ารวม"/"เพิ่มเติม" | 英文"Show X more items" |

### 🎯 识别逻辑对比

| 平台 | 折叠状态识别 | 展开方式 | 延迟时间 |
|------|------------|---------|---------|
| **虾皮** | 包含"สินค้ารวม"文本 | MouseEvent点击 | 400ms |
| **Lazada** | 包含"สินค้ารวม"或"เพิ่มเติม" | click()方法 | 400ms |
| **TikTok** | 包含"Show"+"more"+"item" | click()方法 | 1000ms |

### ⚙️ 技术特点对比

#### 虾皮（Shopee）
- ✅ 结构稳定，很少更新
- ✅ 使用标准事件派发
- ✅ 单一按钮样式
- ⚠️ 需要等待动画完成

#### Lazada
- ✅ 支持两种按钮样式
- ✅ 兼容性好
- ⚠️ 容器层次复杂
- ⚠️ 需要多级查找

#### TikTok Shop
- ⚠️ HTML结构经常变化
- ⚠️ 需要兼容新旧版本
- ✅ 支持动态重新高亮
- ✅ MutationObserver监控
- 🔴 需要更长的延迟时间

### 🔧 维护建议

1. **虾皮平台**：
   - 最稳定，维护成本最低
   - 主要关注泰语文本变化

2. **Lazada平台**：
   - 中等维护成本
   - 需要同时维护两种按钮样式
   - 关注容器结构变化

3. **TikTok平台**：
   - 维护成本最高
   - 需要定期检查选择器有效性
   - 建议每月进行一次结构检查
   - 使用测试脚本`test-tk-buttons.js`验证

**解决方案**：
1. 限制查询范围
2. 使用防抖和节流
3. 批量处理DOM操作
